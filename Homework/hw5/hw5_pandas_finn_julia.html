import re

with open("hw5_raw.txt", "r", encoding="utf-8") as f:
    gilmore_lines = f.readlines()

gilmore_cleaned = [line.strip() for line in gilmore_lines]
gilmore_blanks = [line for line in gilmore_cleaned if line != ""]
gilmore_blanks = [line for line in gilmore_blanks if not line.startswith("[")]
gilmore_blanks = gilmore_blanks[1:]
gilmore_clean = []
speakers_list = []
line_numbers = []
count = 0
for line in gilmore_blanks:
    if "[" in line and "]" in line:
        bracket_one = line.index("[")
        bracket_two = line.index("]")
        line = line[:bracket_one] + line[bracket_two+1:]
    words = re.split(r"[\W]+", line)
    words = [w for w in words if w != ""]
    speakers_list.append(words[0])
    words = words[1:]
    line = " ".join(words)
    gilmore_clean.append(line)
    count += 1
    line_numbers.append(count)

print(gilmore_clean)
